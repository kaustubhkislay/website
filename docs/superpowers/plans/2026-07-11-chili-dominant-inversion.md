```python
import re, sys

def lum(hexc):
    h = hexc.lstrip('#')
    rgb = [int(h[i:i+2], 16) / 255 for i in (0, 2, 4)]
    lin = [c/12.92 if c <= 0.04045 else ((c+0.055)/1.055)**2.4 for c in rgb]
    return 0.2126*lin[0] + 0.7152*lin[1] + 0.0722*lin[2]

def ratio(fg, bg):
    l1, l2 = sorted([lum(fg), lum(bg)], reverse=True)
    return (l1+0.05) / (l2+0.05)

css = open('/Users/kaustubhkislay/website-1/src/app/globals.css').read()
root = css.split(':root')[1].split('}')[0]
tokens = dict(re.findall(r'--([\w-]+):\s*(#[0-9a-fA-F]{6})', root))
bg = tokens['bg']
need_45 = ['text', 'text-muted', 'text-faint', 'text-ghost', 'heading',
           'accent', 'accent-hover', 'tag-research', 'tag-policy',
           'tag-self', 'tag-culture']
failed = False
for name in need_45:
    r = ratio(tokens[name], bg)
    ok = r >= 4.5
    failed |= not ok
    print(f"{'PASS' if ok else 'FAIL'}  --{name} {tokens[name]} on bg: {r:.2f}")
r = ratio(tokens['accent'], bg)
print(f"{'PASS' if r >= 3 else 'FAIL'}  focus ring (accent) >=3:1: {r:.2f}")
sys.exit(1 if failed else 0)
```

### Task 2: Regenerate the Giorno sketch as sand strokes on transparent

**Files:**
- Modify: `public/giorno-sketch.webp` (regenerated in place)

**Interfaces:**
- Consumes: the current sand-baked `public/giorno-sketch.webp` (RGB, background ≈ gray 237).
- Produces: RGBA WebP, same 509×998 dimensions, strokes `#DECDBE`, background alpha 0. `src/app/page.tsx` renders it unchanged.

- [ ] **Step 1: Regenerate the image**

Run with `~/miniconda3/bin/python`:

```python
from PIL import Image

SAND = (0xDE, 0xCD, 0xBE)
im = Image.open('public/giorno-sketch.webp').convert('L')
w, h = im.size

# Background gray level of the sand-baked file (multiply blend left the
# old white bg at the sand luminance, ~237). Pixels at or above bgmax are
# fully transparent; black strokes are fully opaque.
bgmax = 237
out = Image.new('RGBA', (w, h), SAND + (0,))
src = im.load()
dst = out.load()
for y in range(h):
    for x in range(w):
        v = src[x, y]
        a = 0 if v >= bgmax else min(255, round((bgmax - v) * 255 / bgmax))
        dst[x, y] = SAND + (a,)
out.save('public/giorno-sketch.webp', 'WEBP', lossless=True)
print('saved', out.size)
```

- [ ] **Step 2: Pixel-verify the result**

```python
from PIL import Image
im = Image.open('public/giorno-sketch.webp').convert('RGBA')
```
